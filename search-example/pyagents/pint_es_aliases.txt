# pint_es_aliases.txt
# Extensión española para el UnitRegistry estándar de Pint.
#
# Uso:
#   from pint import UnitRegistry
#
#   ureg = UnitRegistry()
#   ureg.load_definitions("pint_es_aliases.txt")
#
# Después:
#   ureg("12 kilómetros")
#   ureg("500 gramos")
#   ureg("3 litros")
#
# NOTA:
# Pint interpreta unidades; no hace NLP sobre texto libre.
# Conviene extraer primero la cantidad (spaCy/reglas) y normalizar
# expresiones como "por", "cuadrados", "cúbicos" o "por ciento".

# LONGITUD
############################

kilometro_es = 1000 * meter = _ = kilometro = kilometros = kilómetro = kilómetros

nanometro_es = 1e-9 * meter = _ = nanometro = nanometros = nanómetro = nanómetros

#
# Pint no interpreta directamente preposiciones/adjetivos del español.
# Antes de crear la Quantity puedes transformar:
#
#   "12 kilómetros por hora" -> "12 kilometro_por_hora"
#   "40 metros cuadrados"    -> "40 metro_cuadrado"
#   "3 metros cúbicos"       -> "3 metro_cubico"
#   "25 grados Celsius"      -> "25 grado_celsius"
#   "15 por ciento"          -> "15 percent"
#   "2 kilovatios hora"      -> "2 kilovatio_hora"
#
# Para un extractor de texto libre:
#
#   texto -> spaCy/reglas -> (valor, unidad textual)
#         -> normalización española -> Pint
